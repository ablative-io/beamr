//! RED: the `&'static` accessor exposure, demonstrated.
//!
//! Each test obtains a slice from one of the four accessor families, forces a
//! GC-triggering allocation while holding it, and reads through it afterwards.
//! Every one of these compiles today because the accessors return `&'static`.
//!
//! Instrument: `cargo +nightly miri test`. Miri reports the ProcBin case as a
//! hard use-after-free; the heap-resident cases are read-back assertions that
//! show the bytes are no longer the bytes.

use beamr::gc;
use beamr::process::Process;
use beamr::term::binary::{Binary, write_binary};
use beamr::term::boxed::{BigInt, ProcBin};
use beamr::term::shared_binary::{SharedBinary, write_proc_bin};

/// ProcBin::as_bytes — the off-heap buffer is released by the collector while
/// the `&'static [u8]` still points at it. Read-after-free.
#[test]
fn procbin_as_bytes_survives_the_arc_it_points_at() {
    let mut process = Process::new(1, 96);
    let shared = SharedBinary::new(vec![0xAB_u8; 128]);

    let words = process
        .heap_mut()
        .alloc_slice_maybe_refcounted(3)
        .expect("proc bin words");
    let term = write_proc_bin(words, &shared).expect("proc bin");

    let bytes: &'static [u8] = ProcBin::new(term).expect("proc bin accessor").as_bytes();
    assert_eq!(bytes[0], 0xAB);

    // The heap word now owns the only strong reference.
    drop(shared);

    // `term` is not a GC root, so the collector releases the Arc under `bytes`.
    gc::collect_minor(&mut process).expect("minor collect");

    // READ AFTER FREE. Miri stops here.
    println!("procbin byte after collect: {:#x}", bytes[0]);
}

/// Binary::as_bytes — inline binary bytes live in heap words. A collection that
/// grows the nursery drops the backing `Vec<u64>` under the `&'static [u8]`.
#[test]
fn inline_binary_as_bytes_survives_the_heap_words_it_points_at() {
    let mut process = Process::new(2, 96);

    let words = process.heap_mut().alloc_slice(3).expect("binary words");
    let term = write_binary(words, b"hello").expect("inline binary");

    let bytes: &'static [u8] = Binary::new(term).expect("binary accessor").as_bytes();
    assert_eq!(bytes, b"hello");

    // Allocation pressure past the nursery: `gc::alloc` collects, then grows,
    // and `Heap::grow_young_to` replaces the backing region outright.
    for _ in 0..64 {
        let _ = gc::alloc(&mut process, 8);
    }

    // READ AFTER FREE / STALE READ.
    println!("inline binary after collect+grow: {:?}", bytes);
    assert_eq!(bytes, b"hello", "the bytes are no longer the bytes");
}

/// BigInt::limbs — the limbs live in heap words, same story.
#[test]
fn bigint_limbs_survive_the_heap_words_they_point_at() {
    let mut process = Process::new(3, 96);

    let words = process.heap_mut().alloc_slice(5).expect("bigint words");
    let term = beamr::term::boxed::write_bigint(words, false, &[0x1122_3344_5566_7788, 0x99])
        .expect("bigint");

    let limbs: &'static [u64] = BigInt::new(term).expect("bigint accessor").limbs();
    assert_eq!(limbs, &[0x1122_3344_5566_7788_u64, 0x99]);

    for _ in 0..64 {
        let _ = gc::alloc(&mut process, 8);
    }

    // READ AFTER FREE / STALE READ.
    println!("bigint limbs after collect+grow: {limbs:?}");
    assert_eq!(
        limbs,
        &[0x1122_3344_5566_7788_u64, 0x99],
        "the limbs are no longer the limbs"
    );
}

/// SubBinary::as_bytes — and with it the private `parent_bytes` helper it calls
/// (`accessors.rs:357`), which is `&'static` for the same reason.
#[test]
fn sub_binary_as_bytes_survives_the_parent_it_views() {
    let mut process = Process::new(4, 96);

    let parent_words = process.heap_mut().alloc_slice(4).expect("parent words");
    let parent = write_binary(parent_words, b"0123456789abcdef").expect("parent binary");
    let sub_words = process.heap_mut().alloc_slice(5).expect("sub words");
    let term = beamr::term::sub_binary::write_sub_binary(sub_words, parent, 4, 6)
        .expect("sub binary");

    let bytes: &'static [u8] = beamr::term::boxed::SubBinary::new(term)
        .expect("sub binary accessor")
        .as_bytes();
    assert_eq!(bytes, b"456789");

    for _ in 0..64 {
        let _ = gc::alloc(&mut process, 8);
    }

    // READ AFTER FREE / STALE READ.
    println!("sub binary after collect+grow: {bytes:?}");
    assert_eq!(bytes, b"456789", "the sub-binary view is no longer the view");
}
